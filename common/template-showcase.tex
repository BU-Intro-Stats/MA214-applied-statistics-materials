\dq{You test positive on a screening test. What is the probability you
actually have the disease?}

\pause

\solnbox{Without the base rate you cannot say. With 1\% prevalence and a 95\%
accurate test, the posterior probability is only about 16\%.}
\end{frame}

\begin{frame}
\frametitle{Formulas and definitions}

\formula{Residual}{The residual is the gap between observed and predicted:
\[ e_i = y_i - \hat{y}_i. \]}

\formula{%
\[ R^2_{adj} = 1 - \pr{\frac{SS_{Error}}{SS_{Total}} \times \frac{n-1}{n-p-1}}. \]}
\end{frame}

\begin{frame}
\frametitle{Caution and key idea}

\caution{A small $p$-value does \hl{not} mean a large or important effect ---
it only speaks to evidence against the null.}

\keyidea{Simulation-based and model-based inference agree when the model
conditions hold; when they do not, prefer the simulation-based answer.}
\end{frame}
